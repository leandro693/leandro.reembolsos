POLICIES ATUAIS DE public.lancamentos
---------------------------------------------------------------------------
lanc_select  SELECT  roles={public}
  USING:
    ((deleted_at IS NULL) AND (usuario_e_owner() OR ((empresa_id IN (
       SELECT empresas_do_usuario())) AND ((user_id = auth.uid()) OR
       (meu_papel(empresa_id) = ANY (ARRAY['financeiro','gestor']))))))

lanc_insert  INSERT  roles={public}
  WITH CHECK:
    ((empresa_id IN (SELECT empresas_do_usuario())) AND (user_id = auth.uid()))

lanc_update  UPDATE  roles={public}
  USING:
    (usuario_e_owner() OR ((empresa_id IN (SELECT empresas_do_usuario())) AND
     ((user_id = auth.uid()) OR (meu_papel(empresa_id) = ANY (ARRAY['financeiro','gestor'])))))
  WITH CHECK: null  (herda o USING)

lanc_delete  DELETE  roles={public}
  USING:
    (usuario_e_owner() OR ((empresa_id IN (SELECT empresas_do_usuario())) AND
     ((user_id = auth.uid()) OR (meu_papel(empresa_id) = 'gestor'))))
